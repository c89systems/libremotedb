cmake_minimum_required(VERSION 3.25)
project(remotedb_sdk_consumer C)
include("${REMOTEDB_SOURCE_ROOT}/cmake/Policy.cmake")
if(METHOD STREQUAL "pkgconfig")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(remotedb REQUIRED IMPORTED_TARGET remotedb remotedbsrv)
  set(libraries PkgConfig::remotedb)
  if(VARIANT STREQUAL "static")
    # pkg-config's static closure is a separate contract from shared linkage.
    set(libraries ${remotedb_STATIC_LIBRARIES})
    if(APPLE)
      foreach(component remotedb remotedbsrv)
        list(TRANSFORM libraries REPLACE "^${component}$" "${SDK_PREFIX}/lib/lib${component}.a")
      endforeach()
    endif()
    if(NOT APPLE)
      add_link_options(-static)
    endif()
  endif()
else()
  find_package(remotedb CONFIG REQUIRED)
  set(libraries remotedb::remotedb_${VARIANT} remotedb::remotedbsrv_${VARIANT})
endif()
add_executable(consumer consumer.c)
add_executable(shipped-example "${SDK_PREFIX}/share/remotedb/examples/stream.c")
foreach(executable consumer shipped-example)
  target_link_libraries(${executable} PRIVATE ${libraries})
  if(METHOD STREQUAL "pkgconfig")
    if(VARIANT STREQUAL "static")
      target_include_directories(${executable} PRIVATE ${remotedb_STATIC_INCLUDE_DIRS})
      target_compile_options(${executable} PRIVATE ${remotedb_STATIC_CFLAGS_OTHER})
      target_link_directories(${executable} PRIVATE ${remotedb_STATIC_LIBRARY_DIRS})
      target_link_options(${executable} PRIVATE ${remotedb_STATIC_LDFLAGS_OTHER})
    else()
      target_include_directories(${executable} PRIVATE ${remotedb_INCLUDE_DIRS})
      target_compile_options(${executable} PRIVATE ${remotedb_CFLAGS_OTHER})
    endif()
  endif()
  remotedb_c89(${executable})
  remotedb_local_runtime(${executable})
endforeach()
foreach(header protocol remotedb server endpoint control service version)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${header}.c" "#include <remotedb/${header}.h>\ntypedef int self_sufficient;\n")
  add_library(header_${header} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/${header}.c")
  target_include_directories(header_${header} PRIVATE "${SDK_PREFIX}/include")
  remotedb_c89(header_${header})
endforeach()
add_executable(private-negative EXCLUDE_FROM_ALL negative.c)
target_link_libraries(private-negative PRIVATE "${SDK_PREFIX}/lib/${SHARED_LIBRARY}")
remotedb_c89(private-negative)
