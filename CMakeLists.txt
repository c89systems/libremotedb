foreach(kind STATIC SHARED)
  if(kind STREQUAL "STATIC" AND NOT REMOTEDB_BUILD_STATIC)
    continue()
  endif()
  if(kind STREQUAL "SHARED" AND NOT REMOTEDB_BUILD_SHARED)
    continue()
  endif()
  string(TOLOWER "${kind}" suffix)
  foreach(component remotedb remotedbsrv)
    if(component STREQUAL "remotedb")
      set(sources ${client})
      set(exports cmake/client.exports)
    else()
      set(sources ${server})
      set(exports cmake/server.exports)
    endif()
    set(target "${component}_${suffix}")
    add_library(${target} ${kind} ${sources})
    target_include_directories(${target} PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
      "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/generated>" "$<INSTALL_INTERFACE:include>")
    target_link_libraries(${target} PRIVATE lonejson::lonejson_static)
    remotedb_c89(${target})
    set_target_properties(${target} PROPERTIES OUTPUT_NAME ${component} EXPORT_NAME "${component}_${suffix}")
    if(kind STREQUAL "SHARED")
      # Exact export allowlists also keep common protocol/IO helpers private.
      file(STRINGS "${exports}" symbols)
      if(APPLE)
        set(control "${CMAKE_CURRENT_BINARY_DIR}/${component}.exports")
        file(WRITE "${control}" "")
        foreach(symbol IN LISTS symbols)
          file(APPEND "${control}" "_${symbol}\n")
        endforeach()
        target_link_options(${target} PRIVATE "LINKER:-exported_symbols_list,${control}")
        set_target_properties(${target} PROPERTIES INSTALL_NAME_DIR "@rpath"
          BUILD_WITH_INSTALL_NAME_DIR TRUE BUILD_WITH_INSTALL_RPATH TRUE)
      else()
        string(REPLACE ";" ";\n" symbol_text "${symbols}")
        set(control "${CMAKE_CURRENT_BINARY_DIR}/${component}.map")
        file(WRITE "${control}" "{ global:\n${symbol_text};\nlocal: *; };\n")
        target_link_options(${target} PRIVATE "LINKER:--version-script=${control}")
      endif()
      set_property(TARGET ${target} APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${exports}")
      set_target_properties(${target} PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0)
    endif()
    if(REMOTEDB_INSTALL)
      install(TARGETS ${target} EXPORT remotedb-targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
    endif()
  endforeach()
endforeach()
